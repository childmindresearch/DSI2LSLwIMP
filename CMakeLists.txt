cmake_minimum_required(VERSION 3.24)
project(dsi2lsl_impedance VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)

set(DSI_API_DIR "${CMAKE_CURRENT_SOURCE_DIR}/DSI_API" CACHE PATH "Vendor DSI SDK folder")
if(NOT IS_DIRECTORY "${DSI_API_DIR}" OR NOT EXISTS "${DSI_API_DIR}/DSI.h" OR NOT EXISTS "${DSI_API_DIR}/DSI_API_Loader.c")
    message(FATAL_ERROR
        "Missing DSI SDK directory, DSI.h, or DSI_API_Loader.c in ${DSI_API_DIR}. "
        "Download the DSI API from Wearable Sensing (https://support.wearablesensing.com/help/downloads/index.html), "
        "extract it, rename the folder to DSI_API and place it in the project root next to CMakeLists.txt, "
        "or pass -DDSI_API_DIR=<path>.")
endif()
option(DSI_FETCH_LIBLSL "Build pinned liblsl 1.17.7 from source (requires internet)" ON)
if(DSI_FETCH_LIBLSL)
    include(FetchContent)
    set(LSL_FRAMEWORK OFF CACHE BOOL "" FORCE)
    set(LSL_BUILD_STATIC OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(liblsl
        URL https://github.com/sccn/liblsl/archive/refs/tags/v1.17.7.tar.gz
        URL_HASH SHA256=19604fb1b30c753457e8bf1430341d84012834c0cf8d2ca921e8151d31220df9
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(liblsl)
else()
    find_package(LSL 1.17 CONFIG REQUIRED)
endif()

if(APPLE)
    if(CMAKE_OSX_ARCHITECTURES MATCHES ";")
        message(FATAL_ERROR "Build one architecture at a time: DSI SDK libraries are not universal")
    endif()
    if(CMAKE_OSX_ARCHITECTURES)
        set(DSI_ARCH "${CMAKE_OSX_ARCHITECTURES}")
    else()
        set(DSI_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    set(DSI_PLATFORM_NAME "Darwin")
    set(DSI_EXTENSION "dylib")
elseif(WIN32)
    set(DSI_PLATFORM_NAME "Windows")
    set(DSI_EXTENSION "dll")
    if(CMAKE_GENERATOR_PLATFORM MATCHES "[Aa][Rr][Mm]64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|arm64|aarch64")
        set(DSI_ARCH "ARM64")
    elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(DSI_ARCH "x64")
    else()
        set(DSI_ARCH "Win32")
    endif()
else()
    set(DSI_PLATFORM_NAME "Linux")
    set(DSI_EXTENSION "so")
    set(DSI_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
    if(DSI_ARCH STREQUAL "aarch64")
        set(DSI_ARCH "arm64")
    endif()
endif()
set(DSI_LIBRARY_NAME "libDSI-${DSI_PLATFORM_NAME}-${DSI_ARCH}.${DSI_EXTENSION}")
set(DSI_LIBRARY "${DSI_API_DIR}/${DSI_LIBRARY_NAME}" CACHE FILEPATH "DSI runtime library for target architecture")
if(NOT EXISTS "${DSI_LIBRARY}")
    message(FATAL_ERROR "Missing ${DSI_LIBRARY}; set DSI_LIBRARY to the correct vendor library")
endif()

add_executable(dsi2lsl dsi2lsl.c "${DSI_API_DIR}/DSI_API_Loader.c")
target_compile_features(dsi2lsl PRIVATE c_std_99)
target_include_directories(dsi2lsl PRIVATE "${DSI_API_DIR}")
target_link_libraries(dsi2lsl PRIVATE LSL::lsl ${CMAKE_DL_LIBS})
if(APPLE)
    target_compile_definitions(dsi2lsl PRIVATE DSI_BUNDLED_LIBRARY="@executable_path/${DSI_LIBRARY_NAME}")
    set_target_properties(dsi2lsl PROPERTIES BUILD_RPATH "@loader_path")
elseif(UNIX)
    target_compile_definitions(dsi2lsl PRIVATE DSI_BUNDLED_LIBRARY="$ORIGIN/${DSI_LIBRARY_NAME}")
    set_target_properties(dsi2lsl PROPERTIES BUILD_RPATH "$ORIGIN")
else()
    target_compile_definitions(dsi2lsl PRIVATE DSI_BUNDLED_LIBRARY="${DSI_LIBRARY_NAME}")
endif()
if(MSVC)
    set_source_files_properties(dsi2lsl.c PROPERTIES COMPILE_OPTIONS /W4)
else()
    set_source_files_properties(dsi2lsl.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Wpedantic")
endif()
add_custom_command(TARGET dsi2lsl POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DSI_LIBRARY}" "$<TARGET_FILE_DIR:dsi2lsl>/${DSI_LIBRARY_NAME}")
if(WIN32)
    add_custom_command(TARGET dsi2lsl POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:LSL::lsl>" "$<TARGET_FILE_DIR:dsi2lsl>")
else()
    add_custom_command(TARGET dsi2lsl POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:LSL::lsl>" "$<TARGET_FILE_DIR:dsi2lsl>/$<TARGET_SONAME_FILE_NAME:LSL::lsl>")
endif()
# Opt-in diagnostic capture tool; separate executable so production dsi2lsl has no overhead.
add_executable(dsi_diagnostics tools/dsi_diagnostics.c "${DSI_API_DIR}/DSI_API_Loader.c")
target_compile_features(dsi_diagnostics PRIVATE c_std_99)
target_include_directories(dsi_diagnostics PRIVATE "${DSI_API_DIR}")
target_link_libraries(dsi_diagnostics PRIVATE LSL::lsl ${CMAKE_DL_LIBS})
if(APPLE)
    target_compile_definitions(dsi_diagnostics PRIVATE DSI_BUNDLED_LIBRARY="@executable_path/${DSI_LIBRARY_NAME}")
    set_target_properties(dsi_diagnostics PROPERTIES BUILD_RPATH "@loader_path")
elseif(UNIX)
    target_compile_definitions(dsi_diagnostics PRIVATE DSI_BUNDLED_LIBRARY="$ORIGIN/${DSI_LIBRARY_NAME}")
    set_target_properties(dsi_diagnostics PROPERTIES BUILD_RPATH "$ORIGIN")
else()
    target_compile_definitions(dsi_diagnostics PRIVATE DSI_BUNDLED_LIBRARY="${DSI_LIBRARY_NAME}")
endif()
if(MSVC)
    set_source_files_properties(tools/dsi_diagnostics.c PROPERTIES COMPILE_OPTIONS /W4)
else()
    set_source_files_properties(tools/dsi_diagnostics.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Wpedantic")
endif()
# Same output directory as dsi2lsl, which already receives the DSI and LSL runtime copies.
add_dependencies(dsi_diagnostics dsi2lsl)
if(BUILD_TESTING)
    add_executable(test_impedance tests/test_impedance.c "${DSI_API_DIR}/DSI_API_Loader.c")
    target_compile_features(test_impedance PRIVATE c_std_99)
    target_include_directories(test_impedance PRIVATE "${DSI_API_DIR}")
    target_link_libraries(test_impedance PRIVATE LSL::lsl ${CMAKE_DL_LIBS})
    add_test(NAME impedance_logic COMMAND test_impedance)
    add_executable(test_dsi_diagnostics tests/test_dsi_diagnostics.c "${DSI_API_DIR}/DSI_API_Loader.c")
    target_compile_features(test_dsi_diagnostics PRIVATE c_std_99)
    target_include_directories(test_dsi_diagnostics PRIVATE "${DSI_API_DIR}")
    target_link_libraries(test_dsi_diagnostics PRIVATE LSL::lsl ${CMAKE_DL_LIBS})
    set(DSI_DIAG_TEST_DIR "${CMAKE_CURRENT_BINARY_DIR}/dsi_diagnostics_test_scratch")
    file(MAKE_DIRECTORY "${DSI_DIAG_TEST_DIR}")
    add_test(NAME dsi_diagnostics_logic COMMAND test_dsi_diagnostics "${DSI_DIAG_TEST_DIR}")
    add_test(NAME dsi_diagnostics_help COMMAND dsi_diagnostics --help)
    add_test(NAME help COMMAND dsi2lsl --help)
    add_test(NAME library_loading COMMAND dsi2lsl --check-libraries)
endif()
